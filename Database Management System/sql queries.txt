CREATE TABLE Country (
    CountryId BIGINT PRIMARY KEY IDENTITY(1,1),
    Country NVARCHAR(255) NOT NULL
);
CREATE TABLE Region (
    RegionId BIGINT PRIMARY KEY IDENTITY(1,1),
    Region NVARCHAR(255) NOT NULL,
    CountryId BIGINT NOT NULL,
    FOREIGN KEY (CountryId) REFERENCES Country(CountryId)
);
CREATE TABLE County (
    CountyId BIGINT PRIMARY KEY IDENTITY(1,1),
    County NVARCHAR(255) NOT NULL,
    RegionId BIGINT NOT NULL,
    FOREIGN KEY (RegionId) REFERENCES Region(RegionId)
);
CREATE TABLE City (
    CityId BIGINT PRIMARY KEY IDENTITY(1,1),
    City NVARCHAR(255) NOT NULL,
    CountyId BIGINT NOT NULL,
    FOREIGN KEY (CountyId) REFERENCES County(CountyId)
);
CREATE TABLE Customer (
    Id BIGINT PRIMARY KEY,
    Gender NVARCHAR(255),
    FirstName NVARCHAR(255),
    LastName NVARCHAR(255),
    DateRegistered DATETIME,
    CityId BIGINT NOT NULL,
    FOREIGN KEY (CityId) REFERENCES City(CityId)
);

CREATE TABLE Order (
    OrderNumber NVARCHAR(50) PRIMARY KEY,
    OrderCreateDate DATETIME,
    OrderStatusCode INT,
    CustomerCityId BIGINT,
    BillingCurrency NVARCHAR(8),
    FOREIGN KEY (CustomerCityId) REFERENCES Customer(CustomerCityId) -- Assuming CustomerCityId exists in Customer table
);
CREATE TABLE OrderItem (
    OrderItemNumber NVARCHAR(32) PRIMARY KEY,
    OrderNumber NVARCHAR(50) NOT NULL,
    ProductGroup NVARCHAR(128) NOT NULL,   -- Foreign key to ProductGroup (from the ProductGroup table)
    ProductCode NVARCHAR(50) NOT NULL,     -- Foreign key to Product (from the Product table)
    VariantCode NVARCHAR(50) NOT NULL,     -- Foreign key to Variant (from the Variant table)
    Quantity INT,
    UnitPrice MONEY,
    LineItemTotal MONEY,
    FOREIGN KEY (OrderNumber) REFERENCES Order(OrderNumber),
    FOREIGN KEY (ProductGroup) REFERENCES ProductGroup(ProductGroup),
    FOREIGN KEY (ProductCode) REFERENCES Product(ProductCode),
    FOREIGN KEY (VariantCode) REFERENCES Variant(VariantCode)
);


INSERT INTO Order (OrderNumber, OrderCreateDate, OrderStatusCode, CustomerCityId, BillingCurrency)
SELECT DISTINCT OrderNumber, OrderCreateDate, OrderStatusCode, CustomerCityId, BillingCurrency
FROM OrderItem;

INSERT INTO OrderItem (
    OrderItemNumber, 
    OrderNumber, 
    ProductGroup, 
    ProductCode, 
    VariantCode, 
    Quantity, 
    UnitPrice, 
    LineItemTotal
)
SELECT 
    oi.OrderItemNumber,
    o.OrderNumber,                   
    pg.ProductGroup,                
    p.ProductCode,                   
    v.VariantCode,                   
    oi.Quantity,
    oi.UnitPrice,
    oi.LineItemTotal
FROM 
    OrderItem oi        
JOIN 
    Order o ON oi.OrderNumber = o.OrderNumber
JOIN 
    ProductGroup pg ON oi.ProductGroup = pg.ProductGroup
JOIN 
    Product p ON oi.ProductCode = p.ProductCode
JOIN 
    Variant v ON oi.VariantCode = v.VariantCode;

CREATE TRIGGER UpdateOrderTotals
ON OrderItemdetails
AFTER INSERT, UPDATE, DELETE
AS
BEGIN
    UPDATE Orders
    SET TotalLineItems = (
            SELECT SUM(Quantity)
            FROM OrderItemdetails
            WHERE OrderItemdetails.OrderNumber = Orders.OrderNumber
        ),
        SavedTotal = (
            SELECT SUM(LineItemTotal)
            FROM OrderItemdetails
            WHERE OrderItemdetails.OrderNumber = Orders.OrderNumber
        )
    WHERE Orders.OrderNumber IN (
        SELECT DISTINCT OrderNumber FROM inserted
        UNION
        SELECT DISTINCT OrderNumber FROM deleted
    );
END;




CREATE PROCEDURE prCreateOrderGroup
    @OrderNumber NVARCHAR(32),
    @OrderCreateDate DATETIME,
    @CustomerCityId INT,
    @OrderStatusCode INT = 0, 
    @BillingCurrency NVARCHAR(8) = 'GBP' 
AS
BEGIN
    -- Start transaction
    BEGIN TRANSACTION;

    BEGIN TRY
        INSERT INTO [Orders] (OrderNumber, OrderCreateDate, OrderStatusCode, CustomerCityId, BillingCurrency, TotalLineItems, SavedTotal)
        VALUES (@OrderNumber, @OrderCreateDate, @OrderStatusCode, @CustomerCityId, @BillingCurrency, 0, 0);

       
        COMMIT TRANSACTION;
        PRINT 'Order created successfully.';
    END TRY

    BEGIN CATCH
        ROLLBACK TRANSACTION;
        PRINT 'Error occurred. Transaction rolled back.';
      
        DECLARE @ErrorMessage NVARCHAR(4000) = ERROR_MESSAGE();
        DECLARE @ErrorSeverity INT = ERROR_SEVERITY();
        DECLARE @ErrorState INT = ERROR_STATE();

        RAISERROR (@ErrorMessage, @ErrorSeverity, @ErrorState);
    END CATCH;
END;





CREATE PROCEDURE prCreateOrderItem
    @OrderNumber NVARCHAR(32),
    @OrderItemNumber NVARCHAR(32),
    @ProductGroup NVARCHAR(128),
    @ProductCode NVARCHAR(255),
    @VariantCode NVARCHAR(255),
    @Quantity INT,
    @UnitPrice MONEY
AS
BEGIN
    -- Calculate the line item total
    DECLARE @LineItemTotal MONEY;
    SET @LineItemTotal = @Quantity * @UnitPrice;

    -- Start transaction
    BEGIN TRANSACTION;

    BEGIN TRY
        -- Check if the order exists
        IF NOT EXISTS (SELECT 1 FROM [Orders] WHERE OrderNumber = @OrderNumber)
        BEGIN
            RAISERROR ('Order does not exist.', 16, 1);
            RETURN;
        END

        -- Insert into the OrderItem table
        INSERT INTO OrderItemdetails (OrderItemNumber, OrderNumber, ProductGroupId, ProductCode, VariantCode, Quantity, UnitPrice, LineItemTotal)
        VALUES (@OrderItemNumber, @OrderNumber, @ProductGroup, @ProductCode, @VariantCode, @Quantity, @UnitPrice, @LineItemTotal);

        -- Update the TotalLineItems and SavedTotal in Order
        UPDATE [Orders]
        SET TotalLineItems = TotalLineItems + @Quantity,
            SavedTotal = SavedTotal + @LineItemTotal
        WHERE OrderNumber = @OrderNumber;

        -- Commit transaction if successful
        COMMIT TRANSACTION;
        PRINT 'Order item created and order updated successfully.';
    END TRY

    BEGIN CATCH
        -- Rollback transaction if an error occurs
        ROLLBACK TRANSACTION;
        PRINT 'Error occurred. Transaction rolled back.';
        
        -- Return error details
        DECLARE @ErrorMessage NVARCHAR(4000) = ERROR_MESSAGE();
        DECLARE @ErrorSeverity INT = ERROR_SEVERITY();
        DECLARE @ErrorState INT = ERROR_STATE();

        RAISERROR (@ErrorMessage, @ErrorSeverity, @ErrorState);
    END CATCH;
END;
